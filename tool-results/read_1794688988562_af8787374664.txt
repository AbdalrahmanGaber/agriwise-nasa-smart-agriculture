     1→     1→/**
     2→     2→ * AgriWise crop suitability engine.
     3→     3→ *
     4→     4→ * Transparent weighted scoring (0-100 per crop):
     5→     5→ *   temperature 30 · water 20 · solar 15 · soil 15 · season 10 · humidity 5 · wind 5
     6→     6→ *
     7→     7→ * Every factor produces an explainable, bilingual justification built from the
     8→     8→ * actual NASA monthly climatology and the farmer's own inputs. When a factor's
     9→     9→ * data is unavailable (e.g. soil type "unknown" or a missing NASA parameter),
    10→    10→ * the factor is marked unavailable, its weight is redistributed across the
    11→    11→ * remaining factors, and the analysis confidence is lowered — no value is ever
    12→    12→ * assumed. Scores are decision support only; they do not guarantee yield or
    13→    13→ * profitability.
    14→    14→ */

    38→    38→/* ---------------- helpers ---------------- */
    39→    39→
    40→    40→const clamp = (v: number, lo = 0, hi = 100) => Math.min(hi, Math.max(lo, v));
    41→    41→const round1 = (v: number) => Math.round(v * 10) / 10;
    42→    42→
    43→    43→/**
    44→    44→ * Piecewise suitability: 0 at/beyond absolute limits, 100 inside the optimum,
    45→    45→ * linear ramps between limit and optimum (10 pts at the absolute limit).
    46→    46→ */
    47→    47→export function rangeFit(
    48→    48→  v: number,
    49→    49→  absMin: number,
    50→    50→  optMin: number,
    51→    51→  optMax: number,
    52→    52→  absMax: number
    53→    53→): number {
    54→    54→  if (v <= absMin || v >= absMax) return 0;
    55→    55→  if (v < optMin) return 10 + (90 * (v - absMin)) / (optMin - absMin);
    56→    56→  if (v > optMax) return 100 - (90 * (v - optMax)) / (absMax - optMax);
    57→    57→  return 100;
    58→    58→}
    59→    59→
    60→    60→const monthRangeLabel = (months: number[]): Bilingual => {
    61→    61→  if (months.length === 0) return L("the growing window", "فترة النمو");
    62→    62→  const fmt = (m: number) => MONTHS_EN[m - 1];
    63→    63→  const fmtAr = (m: number) => MONTHS_AR[m - 1];
    64→    64→  const start = months[0];
    65→    65→  const end = months[months.length - 1];
    66→    66→  if (months.length === 1) return L(fmt(start), fmtAr(start));
    67→    67→  return L(`${fmt(start)}–${fmt(end)}`, `${fmtAr(start)}–${fmtAr(end)}`);
    68→    68→};

    92→    92→/** Growing window: planting month inside the farmer's season when possible,
    93→    93→ *  otherwise the crop's primary documented planting month. */
    94→    94→function pickWindow(
    95→    95→  crop: CropRecord,
    96→    96→  userSeason: FarmInput["season"]
    97→    97→): { startMonth: number; months: number[] } {
    98→    98→  const seasonMonths = SEASON_PLANTING_MONTHS[userSeason];
    99→    99→  const inSeason = crop.plantingMonths.find((m) => seasonMonths.includes(m));
   100→   100→  const start = inSeason ?? crop.plantingMonths[0];
   101→   101→  const midDuration =
   102→   102→    (crop.durationDays[0] + crop.durationDays[1]) / 2;
   103→   103→  const len = Math.max(2, Math.round(midDuration / 30));
   104→   104→  const months = Array.from(
   105→   105→    { length: len },
   106→   106→    (_, i) => ((start - 1 + i) % 12) + 1
   107→   107→  );
   108→   108→  return { startMonth: start, months };
   109→   109→}

   118→   118→function windowMean(
   119→   119→  monthly: MonthlyValue[],
   120→   120→  months: number[],
   121→   121→  key: "t2m" | "t2mMax" | "t2mMin" | "rh2m" | "solar" | "ws10m"
   122→   122→): number | null {
   123→   123→  const vals = months
   124→   124→    .map((m) => monthly[m - 1]?.[key])
   125→   125→    .filter((v): v is number => v !== null && v !== undefined);
   126→   126→  if (vals.length === 0) return null;
   127→   127→  return vals.reduce((a, b) => a + b, 0) / vals.length;
   128→   128→}
   129→   129→
   130→   130→function windowRain(monthly: MonthlyValue[], months: number[]): number | null {
   131→   131→  const vals = months
   132→   132→    .map((m) => monthly[m - 1]?.precip)
   133→   133→    .filter((v): v is number => v !== null && v !== undefined);
   134→   134→  if (vals.length === 0) return null;
   135→   135→  return vals.reduce((a, b) => a + b, 0);
   136→   136→}

   138→   138→function scoreTemperature(
   139→   139→  ctx: Ctx,
   140→   140→  crop: CropRecord,
   141→   141→  months: number[]
   142→   142→): FactorScore {
   143→   143→  const range = monthRangeLabel(months);
   144→   144→  const reasons: Bilingual[] = [];
   145→   145→  const limitations: Bilingual[] = [];
   146→   146→
   147→   147→  const meanT = windowMean(ctx.monthly, months, "t2m");
   148→   148→  if (meanT === null) {
   149→   149→    return {
   150→   150→      key: "temperature",
   151→   151→      weight: BASE_WEIGHTS.temperature,
   152→   152→      effectiveWeight: BASE_WEIGHTS.temperature,
   153→   153→      score: null,
   154→   154→      status: "unavailable",
   155→   155→      reasons: [],
   156→   156→      limitations: [
   157→   157→        L(
   158→   158→          "NASA POWER temperature data unavailable for this location/period.",
   159→   159→          "بيانات درجات الحرارة من NASA POWER غير متاحة لهذا الموقع/الفترة."
   160→   160→        ),
   161→   161→      ],
   162→   162→    };
   163→   163→  }
   164→   164→
   165→   165→  const fits = months.map((m) => {
   166→   166→    const t = ctx.monthly[m - 1]?.t2m;
   167→   167→    return t === null || t === undefined
   168→   168→      ? null
   169→   169→      : rangeFit(t, crop.temp.absMin, crop.temp.optMin, crop.temp.optMax, crop.temp.absMax);
   170→   170→  });
   171→   171→  const validFits = fits.filter((f): f is number => f !== null);
   172→   172→  const meanFit = validFits.length
   173→   173→    ? validFits.reduce((a, b) => a + b, 0) / validFits.length
   174→   174→    : 0;
   175→   175→
   176→   176→  // heat & cold stress from monthly mean maxima/minima
   177→   177→  // 5 penalty points per °C beyond the tolerance, per affected month
   178→   178→  let heatPenalty = 0;
   179→   179→  let coldPenalty = 0;
   180→   180→  for (const m of months) {
   181→   181→    const mv = ctx.monthly[m - 1];
   182→   182→    if (!mv) continue;
   183→   183→    if (mv.t2mMax !== null && mv.t2mMax > crop.temp.absMax) {
   184→   184→      heatPenalty += (mv.t2mMax - crop.temp.absMax) * 5;
   185→   185→      limitations.push(
   186→   186→        L(
   187→   187→          `Daytime maxima in ${MONTHS_EN[m - 1]} average ${round1(mv.t2mMax)}°C, above the ${crop.temp.absMax}°C tolerance limit — heat stress expected.`,
   188→   188→          `متوسط الحرارة القصوى نهارًا في ${MONTHS_AR[m - 1]} ${round1(mv.t2mMax)}°م، أعلى من حد التحمل ${crop.temp.absMax}°م — يُتوقع إجهاد حراري.`
   189→   189→        )
   190→   190→      );
   191→   191→    }
   192→   192→    if (mv.t2mMin !== null && mv.t2mMin < crop.temp.absMin) {
   193→   193→      coldPenalty += (crop.temp.absMin - mv.t2mMin) * 5;
   194→   194→      limitations.push(
   195→   195→        L(
   196→   196→          `Night minima in ${MONTHS_EN[m - 1]} average ${round1(mv.t2mMin)}°C, below the ${crop.temp.absMin}°C minimum — cold stress expected.`,
   197→   197→          `متوسط الحرارة الدنيا ليلًا في ${MONTHS_AR[m - 1]} ${round1(mv.t2mMin)}°م، أقل من الحد الأدنى ${crop.temp.absMin}°م — يُتوقع إجهاد بارد.`
   198→   198→        )
   199→   199→      );
   200→   200→    }
   201→   201→  }
   202→   202→
   203→   203→  const score = clamp(meanFit - heatPenalty - coldPenalty);
   204→   204→
   205→   205→  if (meanT >= crop.temp.optMin && meanT <= crop.temp.optMax) {
   206→   206→    reasons.push(
   207→   207→      L(
   208→   208→        `Mean temperature during ${range.en} is ${round1(meanT)}°C — inside the documented optimum of ${crop.temp.optMin}–${crop.temp.optMax}°C.`,
   209→   209→        `متوسط الحرارة خلال ${range.ar} هو ${round1(meanT)}°م — داخل المدى الأمثل الموثق ${crop.temp.optMin}–${crop.temp.optMax}°م.`
   210→   210→      )
   211→   211→    );
   212→   212→  } else if (meanT > crop.temp.optMax && meanT < crop.temp.absMax) {
   213→   213→    reasons.push(
   214→   214→      L(
   215→   215→        `Mean temperature during ${range.en} is ${round1(meanT)}°C — above the ${crop.temp.optMin}–${crop.temp.optMax}°C optimum but within tolerance.`,
   216→   216→        `متوسط الحرارة خلال ${range.ar} هو ${round1(meanT)}°م — أعلى من المدى الأمثل ${crop.temp.optMin}–${crop.temp.optMax}°م لكنه داخل حد التحمل.`
   217→   217→      )
   218→   218→    );
   219→   219→    limitations.push(
   220→   220→      L(
   221→   221→        `Temperatures run warmer than the documented optimum for most of the window.`,
   222→   222→        `الحرارة أعلى من المدى الأمثل الموثق خلال معظم الفترة.`
   223→   223→      )
   224→   224→    );
   225→   225→  } else if (meanT < crop.temp.optMin && meanT > crop.temp.absMin) {
   226→   226→    reasons.push(
   227→   227→      L(
   228→   228→        `Mean temperature during ${range.en} is ${round1(meanT)}°C — below the ${crop.temp.optMin}–${crop.temp.optMax}°C optimum but within tolerance.`,
   229→   229→        `متوسط الحرارة خلال ${range.ar} هو ${round1(meanT)}°م — أقل من المدى الأمثل ${crop.temp.optMin}–${crop.temp.optMax}°م لكنه داخل حد التحمل.`
   230→   230→      )
   231→   231→    );
   232→   232→    limitations.push(
   233→   233→      L(
   234→   234→        `Temperatures run cooler than the documented optimum for most of the window.`,
   235→   235→        `الحرارة أقل من المدى الأمثل الموثق خلال معظم الفترة.`
   236→   236→      )
   237→   237→    );
   238→   238→  } else {
   239→   239→    limitations.push(
   240→   240→      L(
   241→   241→        `Mean temperature during ${range.en} (${round1(meanT)}°C) is outside the crop's documented tolerance range of ${crop.temp.absMin}–${crop.temp.absMax}°C.`,
   242→   242→        `متوسط الحرارة خلال ${range.ar} (${round1(meanT)}°م) خارج مدى التحمل الموثق للمحصول ${crop.temp.absMin}–${crop.temp.absMax}°م.`
   243→   243→      )
   244→   244→    );
   245→   245→  }
   246→   246→
   247→   247→  return {
   248→   248→    key: "temperature",
   249→   249→    weight: BASE_WEIGHTS.temperature,
   250→   250→    effectiveWeight: BASE_WEIGHTS.temperature,
   251→   251→    score,
   252→   252→    status: heatPenalty + coldPenalty > 15 ? "limited" : "ok",
   253→   253→    reasons,
   254→   254→    limitations,
   255→   255→  };
   256→   256→}

   258→   258→const WATER_SUPPLY: Record<
   259→   259→  CropRecord["water"]["category"],
   260→   260→  Record<FarmInput["waterAvailability"], number>
   261→   261→> = {
   262→   262→  low: { limited: 70, moderate: 95, abundant: 100 },
   263→   263→  medium: { limited: 45, moderate: 80, abundant: 100 },
   264→   264→  high: { limited: 25, moderate: 60, abundant: 95 },
   265→   265→  "very-high": { limited: 10, moderate: 40, abundant: 85 },
   266→   266→};
   267→   267→
   268→   268→function scoreWater(
   269→   269→  ctx: Ctx,
   270→   270→  crop: CropRecord,
   271→   271→  months: number[]
   272→   272→): { factor: FactorScore; gap: number | null } {
   273→   273→  const reasons: Bilingual[] = [];
   274→   274→  const limitations: Bilingual[] = [];
   275→   275→  const [needMin, needMax] = crop.water.seasonNeedMm;
   276→   276→  const needMid = (needMin + needMax) / 2;
   277→   277→  const rain = windowRain(ctx.monthly, months);
   278→   278→  const effRain = rain === null ? 0 : rain;
   279→   279→  const gap = Math.max(0, needMid - effRain);
   280→   280→  const waterLabel = WATER_LABELS[ctx.farm.waterAvailability];
   281→   281→
   282→   282→  reasons.push(
   283→   283→    L(
   284→   284→      `Documented seasonal water requirement: ${needMin}–${needMax} mm per cycle.`,
   285→   285→      `الاحتياج المائي الموسمي الموثق: ${needMin}–${needMax} مم لكل دورة.`
   286→   286→    )
   287→   287→  );
   288→   288→
   289→   289→  if (rain !== null) {
   290→   290→    reasons.push(
   291→   291→      L(
   292→   292→        `Local rainfall during the growing window contributes only about ${round1(effRain)} mm; irrigation must supply the remaining ≈${Math.round(gap)} mm.`,
   293→   293→        `أمطار المنطقة خلال فترة النمو تسهم بنحو ${round1(effRain)} مم فقط؛ وعلى الري تأمين الباقي ≈${Math.round(gap)} مم.`
   294→   294→      )
   295→   295→    );
   296→   296→  } else {
   297→   297→    reasons.push(
   298→   298→      L(
   299→   299→        "NASA rainfall data unavailable for this window; irrigation requirement estimated from crop needs only.",
   300→   300→        "بيانات الأمطار غير متاحة لهذه الفترة؛ قُدِّر الاحتياج من الري وفق احتياج المحصول فقط."
   301→   301→      )
   302→   302→    );
   303→   303→  }
   304→   304→
   305→   305→  const supply = WATER_SUPPLY[crop.water.category][ctx.farm.waterAvailability];
   306→   306→  const rainBonus = rain === null ? 0 : Math.min(10, effRain / 20);
   307→   307→  const score = clamp(supply + rainBonus);
   308→   308→
   309→   309→  if (ctx.farm.waterAvailability === "abundant") {
   310→   310→    reasons.push(
   311→   311→      L(
   312→   312→        `Your reported water availability (${waterLabel.en}) can cover this crop's demand.`,
   313→   313→        `توفر المياه الذي ذكرته (${waterLabel.ar}) يمكنه تغطية احتياج هذا المحصول.`
   314→   314→      )
   315→   315→    );
   316→   316→  } else if (ctx.farm.waterAvailability === "moderate") {
   317→   317→    if (crop.water.category === "high" || crop.water.category === "very-high") {
   318→   318→      limitations.push(
   319→   319→        L(
   320→   320→          `Moderate water availability is tight for a ${crop.water.category === "very-high" ? "very-high" : "high"} requirement crop — plan irrigation carefully.`,
   321→   321→          `التوفر المتوسط للمياه ضيّق لمحصول ذي احتياج ${crop.water.category === "very-high" ? "مرتفع جدًا" : "مرتفع"} — خطّط للري بعناية.`
   322→   322→        )
   323→   323→      );
   324→   324→    } else {
   325→   325→      reasons.push(
   326→   326→        L(
   327→   327→          `Your reported water availability (${waterLabel.en}) is adequate for this requirement level.`,
   328→   328→          `توفر المياه الذي ذكرته (${waterLabel.ar}) كافٍ لمستوى هذا الاحتياج.`
   329→   329→        )
   330→   330→      );
   331→   331→    }
   332→   332→  } else {
   333→   333→    limitations.push(
   334→   334→      L(
   335→   335→        `Limited water availability is a major constraint for this crop (documented need ${needMin}–${needMax} mm).`,
   336→   336→        `محدودية المياه قيد رئيسي لهذا المحصول (الاحتياج الموثق ${needMin}–${needMax} مم).`
   337→   337→      )
   338→   338→    );
   339→   339→  }
   340→   340→
   341→   341→  return {
   342→   342→    factor: {
   343→   343→      key: "water",
   344→   344→      weight: BASE_WEIGHTS.water,
   345→   345→      effectiveWeight: BASE_WEIGHTS.water,
   346→   346→      score,
   347→   347→      status: score < 45 ? "limited" : "ok",
   348→   348→      reasons,
   349→   349→      limitations,
   350→   350→    },
   351→   351→    gap: rain === null ? null : gap,
   352→   352→  };
   353→   353→}

   355→   355→function scoreSolar(
   356→   356→  ctx: Ctx,
   357→   357→  crop: CropRecord,
   358→   358→  months: number[]
   359→   359→): FactorScore {
   360→   360→  const reasons: Bilingual[] = [];
   361→   361→  const limitations: Bilingual[] = [];
   362→   362→  const s = windowMean(ctx.monthly, months, "solar");
   363→   363→  if (s === null) {
   364→   364→    return {
   365→   365→      key: "solar",
   366→   366→      weight: BASE_WEIGHTS.solar,
   367→   367→      effectiveWeight: BASE_WEIGHTS.solar,
   368→   368→      score: null,
   369→   369→      status: "unavailable",
   370→   370→      reasons: [],
   371→   371→      limitations: [
   372→   372→        L(
   373→   373→          "NASA solar radiation data unavailable for this location/period.",
   374→   374→          "بيانات الإشعاع الشمسي من NASA POWER غير متاحة لهذا الموقع/الفترة."
   375→   375→        ),
   376→   376→      ],
   377→   377→    };
   378→   378→  }
   379→   379→  const absMin = crop.solar.optMin - 2;
   380→   380→  const absMax = crop.solar.optMax + 2;
   381→   381→  const score = clamp(rangeFit(s, absMin, crop.solar.optMin, crop.solar.optMax, absMax));
   382→   382→  if (s >= crop.solar.optMin && s <= crop.solar.optMax) {
   383→   383→    reasons.push(
   384→   384→      L(
   385→   385→        `Average solar radiation of ${round1(s)} kWh/m²/day during the window is inside the documented optimum of ${crop.solar.optMin}–${crop.solar.optMax}.`,
   386→   386→        `متوسط الإشعاع الشمسي ${round1(s)} كيلوواط·س/م²/يوم خلال الفترة داخل المدى الأمثل الموثق ${crop.solar.optMin}–${crop.solar.optMax}.`
   387→   387→      )
   388→   388→    );
   389→   389→  } else if (s < crop.solar.optMin) {
   390→   390→    limitations.push(
   391→   391→      L(
   392→   392→        `Average solar radiation of ${round1(s)} kWh/m²/day is below the documented optimum of ${crop.solar.optMin}–${crop.solar.optMax}.`,
   393→   393→        `متوسط الإشعاع الشمسي ${round1(s)} كيلوواط·س/م²/يوم أقل من المدى الأمثل الموثق ${crop.solar.optMin}–${crop.solar.optMax}.`
   394→   394→      )
   395→   395→    );
   396→   396→  } else {
   397→   397→    limitations.push(
   398→   398→      L(
   399→   399→        `Average solar radiation of ${round1(s)} kWh/m²/day exceeds the documented optimum of ${crop.solar.optMin}–${crop.solar.optMax} (higher evaporation and heat load).`,
   400→   400→        `متوسط الإشعاع الشمسي ${round1(s)} كيلوواط·س/م²/يوم أعلى من المدى الأمثل الموثق ${crop.solar.optMin}–${crop.solar.optMax} (تبخّر وحِمل حراري أعلى).`
   401→   401→      )
   402→   402→    );
   403→   403→  }
   404→   404→  return {
   405→   405→    key: "solar",
   406→   406→    weight: BASE_WEIGHTS.solar,
   407→   407→    effectiveWeight: BASE_WEIGHTS.solar,
   408→   408→    score,

   632→   632→export function classify(score: number): Bilingual {
   633→   633→  if (score >= 80) return L("Highly suitable", "ملائم جدًا");
   634→   634→  if (score >= 65) return L("Suitable", "ملائم");
   635→   635→  if (score >= 50) return L("Moderately suitable", "ملائم بشكل متوسط");
   636→   636→  if (score >= 35) return L("Marginally suitable", "ملاءمة محدودة");

   640→   640→function buildSummary(
   641→   641→  crop: CropRecord,
   642→   642→  factors: FactorScore[],
   643→   643→  seasonMatch: boolean
   644→   644→): Bilingual {
   645→   645→  const positives = factors
   646→   646→    .filter((f) => f.score !== null && f.score >= 70 && f.reasons.length > 0)
   647→   647→    .sort((a, b) => b.effectiveWeight - a.effectiveWeight);
   648→   648→  const negatives = factors
   649→   649→    .filter((f) => f.score !== null && f.score < 50 && f.limitations.length > 0)
   650→   650→    .sort((a, b) => b.effectiveWeight - a.effectiveWeight);
   651→   651→
   652→   652→  const posEn = positives.slice(0, 2).map((f) => f.reasons[0].en);
   653→   653→  const posAr = positives.slice(0, 2).map((f) => f.reasons[0].ar);
   654→   654→  const negEn = negatives[0]?.limitations[0]?.en;
   655→   655→  const negAr = negatives[0]?.limitations[0]?.ar;
   656→   656→
   657→   657→  let en = `We recommend growing ${crop.name.en} because `;
   658→   658→  let ar = `نوصي بزراعة ${crop.name.ar} لأن `;
   659→   659→  if (posEn.length === 0) {
   660→   660→    en += "no factor strongly favors it under the current conditions";
   661→   661→    ar += "لا يوجد عامل قوي يرجّحه في الظروف الحالية";
   662→   662→  } else {
   663→   663→    en += posEn.join(", and ");
   664→   664→    ar += posAr.join("، و");
   665→   665→  }
   666→   666→  en += ".";
   667→   667→  ar += ".";
   668→   668→  if (negEn) {
   669→   669→    en += ` Main limitation: ${negEn}`;
   670→   670→    ar += ` القيد الرئيسي: ${negAr}`;
   671→   671→  }
   672→   672→  if (!seasonMatch) {
   673→   673→    const pl = L(
   674→   674→      crop.plantingMonths.map((m) => MONTHS_EN[m - 1]).join(", "),
   675→   675→      crop.plantingMonths.map((m) => MONTHS_AR[m - 1]).join("، ")
   676→   676→    );
   677→   677→    en += ` Note: in Egypt this crop is planted around ${pl.en}, not in your selected season — treat it as planning information for another season.`;
   678→   678→    ar += ` ملاحظة: يُزرع هذا المحصول في مصر حول ${pl.ar} وليس في الموسم الذي اخترته — تعامَل معه كمعلومة تخطيط لموسم آخر.`;
   679→   679→  }
   680→   680→  return L(en, ar);
   681→   681→}

   684→   684→
   685→   685→export function runEngine(
   686→   686→  monthly: MonthlyValue[],
   687→   687→  climateSummary: ClimateSummary,
   688→   688→  farm: FarmInput
   689→   689→): { crops: CropResult[]; engine: EngineInfo } {
   690→   690→  const ctx: Ctx = { monthly, farm };
   691→   691→  const results: CropResult[] = [];
   692→   692→
   693→   693→  for (const crop of CROPS) {
   694→   694→    const { months } = pickWindow(crop, farm.season);
   695→   695→    const seasonMatch = crop.seasons.includes(farm.season);
   696→   696→
   697→   697→    const temperature = scoreTemperature(ctx, crop, months);
   698→   698→    const { factor: water, gap } = scoreWater(ctx, crop, months);
   699→   699→    const solar = scoreSolar(ctx, crop, months);
   700→   700→    const soil = scoreSoil(ctx, crop);
   701→   701→    const season = scoreSeason(ctx, crop);
   702→   702→    const humidity = scoreHumidity(ctx, crop, months);
   703→   703→    const wind = scoreWind(ctx, crop, months);
   704→   704→
   705→   705→    const factors = [temperature, water, solar, soil, season, humidity, wind];
   706→   706→
   707→   707→    // redistribute weight of unavailable factors
   708→   708→    const available = factors.filter((f) => f.score !== null);
   709→   709→    const totalBase = available.reduce((a, f) => a + f.weight, 0);
   710→   710→    for (const f of factors) {
   711→   711→      f.effectiveWeight =
   712→   712→        f.score === null ? 0 : Math.round((f.weight / totalBase) * 1000) / 10;
   713→   713→    }
   714→   714→
   715→   715→    const score =
   716→   716→      available.length === 0
   717→   717→        ? 0
   718→   718→        : Math.round(
   719→   719→            available.reduce((a, f) => a + (f.score as number) * f.weight, 0) /
   720→   720→              totalBase
   721→   721→          );
   722→   722→
   723→   723→    results.push({
   724→   724→      cropId: crop.id,
   725→   725→      score,
   726→   726→      classification: classify(score),
   727→   727→      rank: 0,
   728→   728→      seasonMatch,
   729→   729→      factors,
   730→   730→      windowMonths: months,
   731→   731→      irrigationGapMm: gap,
   732→   732→      summary: buildSummary(crop, factors, seasonMatch),
   733→   733→    });
   734→   734→  }
   735→   735→
   736→   736→  // Crops plantable in the farmer's selected season rank first,
   737→   737→  // then off-season crops (clearly labeled in the UI as planning info).
   738→   738→  results.sort((a, b) =>
   739→   739→    a.seasonMatch === b.seasonMatch ? b.score - a.score : a.seasonMatch ? -1 : 1
   740→   740→  );
   741→   741→  results.forEach((r, i) => (r.rank = i + 1));
   742→   742→
   743→   743→  // confidence
   744→   744→  const unavailableClimate = results[0]?.factors.filter(
   745→   745→    (f) => f.status === "unavailable" && f.key !== "soil"
   746→   746→  );
   747→   747→  const soilUnknown = farm.soilType === "unknown";
   748→   748→  const coverage = climateSummary.coverageOverall;
   749→   749→  let confidence: EngineInfo["confidence"] = "high";
   750→   750→  const notes: Bilingual[] = [];
   751→   751→  if (coverage < 0.85) {
   752→   752→    confidence = "low";
   753→   753→    notes.push(
   754→   754→      L(
   755→   755→        `NASA data coverage for the analyzed period is only ${Math.round(coverage * 100)}%.`,
   756→   756→        `تغطية بيانات NASA للفترة المحللة ${Math.round(coverage * 100)}% فقط.`
   757→   757→      )
   758→   758→    );
   759→   759→  } else if (coverage < 0.95 || soilUnknown || (unavailableClimate?.length ?? 0) > 0) {
   760→   760→    confidence = "medium";
   761→   761→  }
   762→   762→  if (soilUnknown) {
   763→   763→    notes.push(
   764→   764→      L(
   765→   765→        "Soil type was not provided, so the soil factor was excluded from scoring.",
   766→   766→        "لم يُحدَّد نوع التربة، لذا استُبعد عامل التربة من التقييم."
   767→   767→      )
   768→   768→    );
   769→   769→  }
   770→   770→  if ((unavailableClimate?.length ?? 0) > 0) {
   771→   771→    notes.push(
   772→   772→      L(
   773→   773→        "Some NASA parameters were unavailable and their factors were excluded.",
   774→   774→        "بعض متغيرات NASA لم تكن متاحة واستُبعدت عواملها."
   775→   775→      )
   776→   776→    );
   777→   777→  }
   778→   778→  if (notes.length === 0) {
   779→   779→    notes.push(
   780→   780→      L(
   781→   781→        "All factors were evaluated with complete NASA POWER data and farm inputs.",
   782→   782→        "قُيِّمت جميع العوامل ببيانات NASA POWER كاملة ومدخلات المزرعة."
   783→   783→      )
   784→   784→    );
   785→   785→  }
   786→   786→
   787→   787→  const engine: EngineInfo = {
   788→   788→    weights: FACTOR_ORDER.map((key) => ({ key, weight: BASE_WEIGHTS[key] })),
   789→   789→    methodology: L(
   790→   790→      "Each factor is scored 0-100 by comparing NASA POWER monthly climatology and your farm inputs with the documented requirements of each crop (FAO EcoCrop, FAO I&D Paper 33, Egyptian MALR extension data). The final score is the weighted average: temperature 30%, water 20%, solar radiation 15%, soil 15%, season 10%, humidity 5%, wind 5%. Unavailable factors are excluded and their weight redistributed. Scores support decisions; they do not guarantee yield or profit.",
   791→   791→      "يُحسب لكل عامل درجة من 0 إلى 100 بمقارنة المناخ الشهري من NASA POWER ومدخلات مزرعتك بالاحتياجات الموثقة لكل محصول (FAO EcoCrop وورقة FAO رقم 33 وإرشادات وزارة الزراعة المصرية). الدرجة النهائية هي المتوسط المرجّح: الحرارة 30%، والماء 20%، والإشعاع الشمسي 15%، والتربة 15%، والموسم 10%، والرطوبة 5%، والرياح 5%. العوامل غير المتاحة تُستبعد ويُعاد توزيع وزنها. الدرجات لدعم القرار ولا تضمن المحصول أو الربح."
   792→   792→    ),
   793→   793→    confidence,
   794→   794→    confidenceNote: L(notes.map((n) => n.en).join(" "), notes.map((n) => n.ar).join(" ")),
   795→   795→  };
   796→   796→
   797→   797→  return { crops: results, engine };
   798→   798→}
